#Radix

uart_println_str("Testing radix");

u64 tree_root = kernel_radix_create_tree(4);

uart_print_str("table made at : ");
uart_println_u64_hex(tree_root);

uart_println_str("Setting values");

for (u64 i = 0; i<100000; i++) {
    u64 *child_pointer = (u64 *)kernel_allocator_acquire(8);
    *child_pointer = (i*i)+68359;
    kernel_radix_create_child(tree_root, i, (u64)child_pointer, 7, 3);
}

uart_println_str("printing out values");

for (u64 i = 0; i<100000; i++) {
    u64 *child_pointer = (u64 *)kernel_allocator_acquire(8);
    *child_pointer = i;
    u64 *child = (u64 *)kernel_radix_get_child(tree_root, i, 7, 3);
    if (!child) {
        uart_println_str("0");
    }else{
        if (*child != (i*i)+68359) {
            uart_print_u64_hex((u64)child);
            uart_print_str(" : ");
            uart_println_u64(*child);
        }
    }
}
